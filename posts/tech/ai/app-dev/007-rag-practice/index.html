概述

本章将上一章的RAG理论落地为完整项目，使用DeepSeek + FAISS搭建一个本地知识库问答系统。项目覆盖PDF文本提取、文档分块、向量索引构建、语义搜索与问答链的全流程开发，并介绍GraphRAG的全局-局部搜索机制和Qwen-Agent的多级RAG应用。

一、项目架构设计

1.1 整体架构

PDF文档 → 文本提取 → 文档分块 → Embedding → FAISS索引
                                                    ↓
用户提问 → Query Embedding → 向量检索 → 上下文组装 → LLM生成 → 答案

1.2 技术栈

组件选型说明
LLMDeepSeek-R1 (Ollama)本地推理，数据安全
Embeddingtext-embedding-v3阿里云DashScope
向量数据库FAISS轻量高效
PDF解析PyMuPDF高质量文本提取
框架LangChain编排与集成

2.1 使用PyMuPDF提取文本

 2import logging

 7logger = logging.getLogger(__name__)

10@dataclass
11class DocumentChunk:
12    content: str
13    source: str
14    page: int
15    chunk_id: int

17
18class PDFExtractor:
19    def __init__(self, chunk_size: int = 500, chunk_overlap: int = 50):
20        self.chunk_size = chunk_size
21        self.chunk_overlap = chunk_overlap
22
23    def extract_text(self, pdf_path: str) -> list[dict]:
24        doc = fitz.open(pdf_path)
25        pages = []
26        for page_num in range(len(doc)):
27            page = doc[page_num]
28            text = page.get_text("text")
29            if text.strip():
30                pages.append({
31                    "content": text.strip(),
32                    "source": Path(pdf_path).name,
33                    "page": page_num + 1
34                })
35        doc.close()
36        logger.info(f"提取完成: {pdf_path}, 共 {len(pages)} 页")
37        return pages
38
39    def extract_and_chunk(self, pdf_path: str) -> list[DocumentChunk]:
40        pages = self.extract_text(pdf_path)
41        all_chunks = []
42        chunk_id = 0
43
44        for page_info in pages:
45            chunks = self._recursive_chunk(page_info["content"])
46            for chunk in chunks:
47                all_chunks.append(DocumentChunk(
48                    content=chunk,
49                    source=page_info["source"],
50                    page=page_info["page"],
51                    chunk_id=chunk_id
52                ))
53                chunk_id += 1
54
55        logger.info(f"分块完成: {pdf_path}, 共 {len(all_chunks)} 个块")
56        return all_chunks
57
58    def _recursive_chunk(self, text: str) -> list[str]:
59        if len(text) <= self.chunk_size:
60            return [text]
61
62        separators = ["\n\n", "\n", "。", "！", "？", ".", " "]
63        for sep in separators:
64            if sep in text:
65                parts = text.split(sep)
66                chunks = []
67                current = ""
68                for part in parts:
69                    if len(current) + len(part) > self.chunk_size and current:
70                        chunks.append(current.strip())
71                        overlap = current[-self.chunk_overlap:] if self.chunk_overlap > 0 else ""
72                        current = overlap + part + sep
73                    else:
74                        current += part + sep
75                if current.strip():
76                    chunks.append(current.strip())
77                return chunks
78
79        return [text[i:i + self.chunk_size] for i in range(0, len(text), self.chunk_size - self.chunk_overlap)]
80
81
82def batch_extract(pdf_dir: str) -> list[DocumentChunk]:
83    extractor = PDFExtractor(chunk_size=500, chunk_overlap=50)
84    all_chunks = []
85    pdf_path = Path(pdf_dir)
86
87    for file in pdf_path.glob("*.pdf"):
88        chunks = extractor.extract_and_chunk(str(file))
89        all_chunks.extend(chunks)
90
91    logger.info(f"批量提取完成，共 {len(all_chunks)} 个文档块")
92    return all_chunks

三、向量索引构建

3.1 Embedding生成与FAISS索引

 1import dashscope
 2import numpy as np
 3import faiss
 4import json
 5import pickle
 6import logging
 7import os
 8from typing import Optional
 9
10logging.basicConfig(level=logging.INFO)
11logger = logging.getLogger(__name__)
12
13dashscope.api_key = os.getenv("DASHSCOPE_API_KEY", "sk-xxx")
14
15
16class KnowledgeBase:
17    def __init__(self, dimension: int = 1024):
18        self.dimension = dimension
19        self.index = faiss.IndexFlatIP(dimension)
20        self.chunks: list[DocumentChunk] = []
21        self.chunk_embeddings: list[list[float]] = []
22
23    def _get_embeddings_batch(self, texts: list[str], batch_size: int = 25) -> list[list[float]]:
24        all_embeddings = []
25        for i in range(0, len(texts), batch_size):
26            batch = texts[i:i + batch_size]
27            try:
28                resp = dashscope.TextEmbedding.call(
29                    model="text-embedding-v3",
30                    input=batch
31                )
32                if resp.status_code == 200:
33                    batch_embs = [item["embedding"] for item in resp.output["embeddings"]]
34                    all_embeddings.extend(batch_embs)
35                    logger.info(f"Embedding批次 {i // batch_size + 1} 完成，数量: {len(batch)}")
36                else:
37                    logger.error(f"Embedding失败: {resp.code} - {resp.message}")
38            except Exception as e:
39                logger.error(f"Embedding异常: {e}")
40        return all_embeddings
41
42    def build_index(self, chunks: list[DocumentChunk]):
43        self.chunks = chunks
44        texts = [chunk.content for chunk in chunks]
45
46        logger.info(f"开始构建索引，文档块数: {len(texts)}")
47        embeddings = self._get_embeddings_batch(texts)
48
49        if len(embeddings) != len(chunks):
50            logger.error(f"Embedding数量不匹配: {len(embeddings)} vs {len(chunks)}")
51            return
52
53        self.chunk_embeddings = embeddings
54        vectors = np.array(embeddings, dtype=np.float32)
55        faiss.normalize_L2(vectors)
56
57        self.index = faiss.IndexFlatIP(self.dimension)
58        self.index.add(vectors)
59        logger.info(f"索引构建完成，总文档块: {self.index.ntotal}")
60
61    def search(self, query: str, top_k: int = 5) -> list[tuple[DocumentChunk, float]]:
62        query_emb = self._get_embeddings_batch([query])
63        if not query_emb:
64            return []
65
66        query_vec = np.array(query_emb, dtype=np.float32)
67        faiss.normalize_L2(query_vec)
68
69        scores, indices = self.index.search(query_vec, top_k)
70
71        results = []
72        for score, idx in zip(scores[0], indices[0]):
73            if 0 <= idx < len(self.chunks):
74                results.append((self.chunks[idx], float(score)))
75        return results
76
77    def save(self, index_path: str = "kb_index.bin", meta_path: str = "kb_meta.pkl"):
78        faiss.write_index(self.index, index_path)
79        with open(meta_path, "wb") as f:
80            pickle.dump({
81                "chunks": self.chunks,
82                "dimension": self.dimension
83            }, f)
84        logger.info(f"知识库已保存: {index_path}, {meta_path}")
85
86    def load(self, index_path: str = "kb_index.bin", meta_path: str = "kb_meta.pkl"):
87        self.index = faiss.read_index(index_path)
88        with open(meta_path, "rb") as f:
89            meta = pickle.load(f)
90            self.chunks = meta["chunks"]
91            self.dimension = meta["dimension"]
92        logger.info(f"知识库已加载，文档块数: {self.index.ntotal}")

五、GraphRAG全局-局部搜索

5.1 GraphRAG在知识库中的应用

传统RAG擅长回答具体问题，但面对"总结所有文档的核心观点"这类全局性问题时效果不佳。GraphRAG通过构建知识图谱，实现双层搜索。

  1from dataclasses import dataclass, field
  2from collections import defaultdict
  3import json
  4import logging
  5
  6logging.basicConfig(level=logging.INFO)
  7logger = logging.getLogger(__name__)
  8
  9
 10@dataclass
 11class KnowledgeEntity:
 12    name: str
 13    entity_type: str
 14    description: str
 15    source_docs: list[str] = field(default_factory=list)
 16
 17
 18@dataclass
 19class KnowledgeRelation:
 20    source_entity: str
 21    target_entity: str
 22    relation: str
 23    description: str
 24
 25
 26@dataclass
 27class CommunitySummary:
 28    community_id: str
 29    entities: list[str]
 30    summary: str
 31    level: int
 32
 33
 34class GraphRAGBuilder:
 35    def __init__(self, llm_call_fn):
 36        self.llm_call = llm_call_fn
 37        self.entities: list[KnowledgeEntity] = []
 38        self.relations: list[KnowledgeRelation] = []
 39        self.communities: list[CommunitySummary] = []
 40        self.entity_index: dict[str, list[str]] = defaultdict(list)
 41
 42    def extract_from_chunk(self, chunk: DocumentChunk) -> None:
 43        prompt = f"""从以下文本中提取实体和关系，以JSON格式输出。
 44
 45文本：{chunk.content}
 46
 47输出格式：
 48{{
 49    "entities": [{{"name": "名称", "type": "类型", "description": "描述"}}],
 50    "relations": [{{"source": "源实体", "target": "目标实体", "relation": "关系", "description": "描述"}}]
 51}}"""
 52
 53        response = self.llm_call([{"role": "user", "content": prompt}])
 54        try:
 55            json_str = response.strip()
 56            if "```json" in json_str:
 57                json_str = json_str.split("```json")[1].split("```")[0].strip()
 58            data = json.loads(json_str)
 59
 60            for e in data.get("entities", []):
 61                entity = KnowledgeEntity(
 62                    name=e["name"],
 63                    entity_type=e["type"],
 64                    description=e["description"],
 65                    source_docs=[chunk.source]
 66                )
 67                self.entities.append(entity)
 68                self.entity_index[e["name"]].append(chunk.source)
 69
 70            for r in data.get("relations", []):
 71                relation = KnowledgeRelation(
 72                    source_entity=r["source"],
 73                    target_entity=r["target"],
 74                    relation=r["relation"],
 75                    description=r["description"]
 76                )
 77                self.relations.append(relation)
 78
 79        except (json.JSONDecodeError, KeyError) as e:
 80            logger.warning(f"实体提取失败: {e}")
 81
 82    def build_communities(self) -> None:
 83        adjacency = defaultdict(set)
 84        for r in self.relations:
 85            adjacency[r.source_entity].add(r.target_entity)
 86            adjacency[r.target_entity].add(r.source_entity)
 87
 88        for e in self.entities:
 89            if e.name not in adjacency:
 90                adjacency[e.name] = set()
 91
 92        visited = set()
 93        community_id = 0
 94
 95        for entity_name in adjacency:
 96            if entity_name not in visited:
 97                component = self._bfs(entity_name, adjacency, visited)
 98                entity_descs = []
 99                for name in component:
100                    matching = [e for e in self.entities if e.name == name]
101                    if matching:
102                        entity_descs.append(f"- {name}({matching[0].entity_type}): {matching[0].description}")
103
104                summary = self._generate_community_summary(component, entity_descs)
105                self.communities.append(CommunitySummary(
106                    community_id=f"c_{community_id}",
107                    entities=component,
108                    summary=summary,
109                    level=0
110                ))
111                community_id += 1
112
113        logger.info(f"社区构建完成，共 {len(self.communities)} 个社区")
114
115    def _bfs(self, start: str, adjacency: dict, visited: set) -> list[str]:
116        queue = [start]
117        component = []
118        while queue:
119            node = queue.pop(0)
120            if node not in visited:
121                visited.add(node)
122                component.append(node)
123                for neighbor in adjacency.get(node, set()):
124                    if neighbor not in visited:
125                        queue.append(neighbor)
126        return component
127
128    def _generate_community_summary(self, entities: list[str], descriptions: list[str]) -> str:
129        desc_text = "\n".join(descriptions[:20])
130        prompt = f"""请为以下一组相关实体生成简洁的社区摘要：
131
132实体列表：
133{desc_text}
134
135社区摘要："""
136        return self.llm_call([{"role": "user", "content": prompt}])
137
138    def global_search(self, query: str) -> str:
139        community_contexts = []
140        for c in self.communities:
141            community_contexts.append(f"社区 {c.community_id}:\n{c.summary}")
142
143        context = "\n\n".join(community_contexts[:10])
144        prompt = f"""基于以下社区摘要回答全局性问题。
145
146社区摘要：
147{context}
148
149问题：{query}
150
151请综合所有社区信息给出全面回答："""
152        return self.llm_call([{"role": "user", "content": prompt}])
153
154    def local_search(self, query: str, vector_store: KnowledgeBase, top_k: int = 5) -> str:
155        results = vector_store.search(query, top_k=top_k)
156        context = "\n\n".join([chunk.content for chunk, _ in results])
157        prompt = f"""基于以下文档片段回答问题。
158
159文档内容：
160{context}
161
162问题：{query}
163
164回答："""
165        return self.llm_call([{"role": "user", "content": prompt}])

六、Qwen-Agent多级RAG

6.1 多级RAG架构

Qwen-Agent提供了多级RAG能力，支持从简单检索到复杂推理的渐进式检索策略。

 1import dashscope
 2import logging
 3import os
 4
 5logging.basicConfig(level=logging.INFO)
 6logger = logging.getLogger(__name__)
 7
 8dashscope.api_key = os.getenv("DASHSCOPE_API_KEY", "sk-xxx")
 9
10
11class MultiLevelRAG:
12    def __init__(self, knowledge_base: KnowledgeBase):
13        self.kb = knowledge_base
14
15    def _call_qwen(self, messages: list[dict], model: str = "qwen-turbo") -> str:
16        try:
17            response = dashscope.Generation.call(
18                model=model,
19                messages=messages,
20                result_format="message"
21            )
22            if response.status_code == 200:
23                return response.output.choices[0].message.content
24            return ""
25        except Exception as e:
26            logger.error(f"Qwen调用失败: {e}")
27            return ""
28
29    def level1_direct_search(self, query: str, top_k: int = 3) -> str:
30        results = self.kb.search(query, top_k=top_k)
31        context = "\n".join([chunk.content for chunk, _ in results])
32        messages = [
33            {"role": "system", "content": "基于上下文回答问题，不要编造信息。"},
34            {"role": "user", "content": f"上下文：{context}\n\n问题：{query}"}
35        ]
36        return self._call_qwen(messages)
37
38    def level2_query_rewrite_search(self, query: str) -> str:
39        rewrite_messages = [
40            {"role": "user", "content": f"将以下问题改写为更适合检索的查询：\n{query}"}
41        ]
42        rewritten = self._call_qwen(rewrite_messages)
43        logger.info(f"查询改写: {query} → {rewritten}")
44
45        results = self.kb.search(rewritten, top_k=5)
46        context = "\n".join([chunk.content for chunk, _ in results])
47        messages = [
48            {"role": "system", "content": "基于上下文回答问题，不要编造信息。"},
49            {"role": "user", "content": f"上下文：{context}\n\n原始问题：{query}"}
50        ]
51        return self._call_qwen(messages)
52
53    def level3_multi_step_search(self, query: str) -> str:
54        decompose_messages = [
55            {"role": "user", "content": f"将以下复杂问题拆解为2-3个子问题：\n{query}"}
56        ]
57        sub_questions = self._call_qwen(decompose_messages)
58        logger.info(f"问题拆解: {sub_questions}")
59
60        all_contexts = []
61        for line in sub_questions.strip().split("\n"):
62            line = line.strip()
63            if line and (line[0].isdigit() or line.startswith("-")):
64                sub_q = line.lstrip("0123456789.-) ")
65                results = self.kb.search(sub_q, top_k=2)
66                for chunk, score in results:
67                    all_contexts.append(f"[相关度:{score:.3f}] {chunk.content}")
68
69        context = "\n\n".join(all_contexts[:10])
70        messages = [
71            {"role": "system", "content": "基于上下文综合回答问题，不要编造信息。"},
72            {"role": "user", "content": f"上下文：{context}\n\n问题：{query}"}
73        ]
74        return self._call_qwen(messages)
75
76    def smart_search(self, query: str) -> str:
77        classify_messages = [
78            {"role": "user", "content": f"判断以下问题的复杂度，只输出 simple/medium/complex：\n{query}"}
79        ]
80        complexity = self._call_qwen(classify_messages).strip().lower()
81        logger.info(f"问题复杂度: {complexity}")
82
83        if complexity == "simple":
84            return self.level1_direct_search(query)
85        elif complexity == "medium":
86            return self.level2_query_rewrite_search(query)
87        else:
88            return self.level3_multi_step_search(query)

七、RAG效果优化实践

7.1 常见问题与优化方向

问题原因优化方案
检索不到相关文档分块过大/查询不精确优化分块策略 + 查询改写
回答包含无关信息检索结果噪声多增加重排序 + 上下文压缩

 64. 对于数值类问题，精确引用原文数据
 75. 对于概念类问题，综合多个相关片段给出完整解释
 8
 9回答格式：
10- 直接回答问题
11- 列出关键依据
12- 标注信息来源"""

总结

本章通过完整项目实践了RAG技术的落地：

  1. PDF文本提取：使用PyMuPDF高质量提取文本，递归分块保证语义完整
  2. 向量索引构建：FAISS + DashScope Embedding构建高效语义检索
  3. 问答链实现：DeepSeek本地推理 + 上下文注入，实现安全可控的知识问答
  4. GraphRAG：知识图谱 + 社区摘要，解决全局性问题的推理需求
  5. 多级RAG：根据问题复杂度自适应选择检索策略，平衡效果与效率

核心经验：RAG的效果取决于"检索质量"和"生成质量"两个维度，检索是基础，生成是关键，两者需要协同优化。
